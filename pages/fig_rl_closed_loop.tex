% Background boxes

  \node[draw=blue!45!black, rounded corners=10pt, inner sep=0.42cm, fit=(lactor) (state) (actor) (critic) (lcritic)] (agentbox) {};
  \node[draw=green!40!black, rounded corners=10pt, inner sep=0.34cm, fit=(env) (reward)] (envbox) {};
